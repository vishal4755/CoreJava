         *** MySQL Queries ***



 1. Create Database

CREATE DATABASE college_db;

 2. Use Database

USE college_db;

 Table Queries

 3. Create Table

CREATE TABLE student (
    id INT PRIMARY KEY,
    name VARCHAR(50),
    age INT,
    city VARCHAR(50)
);

4. Show Tables

SHOW TABLES;


 5. Describe Table

DESC student;

 6. Rename Table

RENAME TABLE student TO students;


7. Delete Table

DROP TABLE students;

 Select Queries

 WHERE Clause

13. Equal Condition

SELECT * FROM student WHERE city='Pune';

 14. AND Condition

SELECT * FROM student WHERE age > 20 AND city='Pune';

 15. OR Condition


SELECT * FROM student WHERE city='Pune' OR city='Mumbai';

 16. NOT Condition


SELECT * FROM student WHERE NOT city='Pune';

 Operators

 17. BETWEEN


SELECT * FROM student WHERE age BETWEEN 20 AND 25;


 18. IN Operator

 19. LIKE Operator

20. Wildcards

 GROUP BY & HAVING

 35. GROUP BY

SELECT city, COUNT(*)
FROM student
GROUP BY city;


36. HAVING

SELECT city, COUNT(*)
FROM student
GROUP BY city
HAVING COUNT(*) > 1;

Joins

 37. INNER JOIN

SELECT s.name, c.course_name
FROM student s
INNER JOIN course c
ON s.id = c.student_id;

 38. LEFT JOIN

SELECT s.name, c.course_name
FROM student s
LEFT JOIN course c
ON s.id = c.student_id;


 39. RIGHT JOIN

SELECT s.name, c.course_name
FROM student s
RIGHT JOIN course c
ON s.id = c.student_id;

 40. FULL JOIN (MySQL Alternative)

SELECT * FROM student
LEFT JOIN course
ON student.id = course.student_id

UNION

SELECT * FROM student
RIGHT JOIN course
ON student.id = course.student_id;
